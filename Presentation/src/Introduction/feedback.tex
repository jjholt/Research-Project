\begin{frame}
    \frametitle{Vibrotactile (haptic) feedback}
    \begin{itemize}
        \item Use vibrations for feedback
        \item Move away from relying on secondary cues: e.g. sounds
        \item Distinction of discrete patterns: ``feeling''
        \item Clinically relevant frequency: up to about \SI{250}{Hz}
    \end{itemize}
