\begin{frame}
    \frametitle{Simulation setup}
    Change one variable at a time
    \begin{itemize}
        \item Load: $ A \cos \left( 2\pi f t \right) $
        \begin{itemize}
            \item A: \SI{200}{\micro\meter}
            \item f: \SI{100}{Hz}
        \end{itemize}
        \item Period: $ 6/f $ seconds. e.g., \SI{60}{\milli\second} for \SI{100}{Hz}
        \item Time-increment: $ 1/(30f) $ seconds. e.g., \SI{0.3}{\milli\second} for \SI{100}{Hz}
        \item Position of load: \SI{1}{mm} from base of spigot.
    \end{itemize}
    
\end{frame}
